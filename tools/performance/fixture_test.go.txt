// Diagnostic limits, not release budgets. In particular neither the 3-CPU-second
// settled target nor the 10-second append target is a correctness test gate.
const perfCPU = 30.0
const perfFootprint = 256 << 20
const perfSpool = 64 << 20

// Attempt counts, including failed attempts. Read bytes count bytes actually read;
// put/CAS bytes count offered bytes. Sync is forwarded: never weaken Local durability.
type perfBackend struct {
	publication.Backend
	mu     sync.Mutex
	counts map[string]int64
}

func (b *perfBackend) add(k string, n int64) { b.mu.Lock(); b.counts[k] += n; b.mu.Unlock() }
func (b *perfBackend) snapshot() map[string]int64 {
	b.mu.Lock()
	defer b.mu.Unlock()
	m := map[string]int64{}
	for k, v := range b.counts {
		m[k] = v
	}
	return m
}

type perfRead struct {
	io.ReadCloser
	b *perfBackend
}

func (r perfRead) Read(p []byte) (int, error) {
	n, e := r.ReadCloser.Read(p)
	r.b.add("backend_read_bytes", int64(n))
	return n, e
}
func (b *perfBackend) Open(c context.Context, k string) (io.ReadCloser, error) {
	b.add("backend_open", 1)
	r, e := b.Backend.Open(c, k)
	if e != nil {
		return nil, e
	}
	return perfRead{r, b}, nil
}
func (b *perfBackend) Current(c context.Context, k string) ([]byte, string, error) {
	b.add("backend_current", 1)
	p, t, e := b.Backend.Current(c, k)
	b.add("backend_read_bytes", int64(len(p)))
	return p, t, e
}
func (b *perfBackend) PutFile(c context.Context, k, p string) error {
	b.add("backend_put", 1)
	s, e := os.Stat(p)
	if e != nil {
		return e
	}
	b.add("backend_put_bytes", s.Size())
	return b.Backend.PutFile(c, k, p)
}
func (b *perfBackend) CompareAndSwap(c context.Context, k, t string, p []byte) error {
	b.add("backend_cas", 1)
	b.add("backend_cas_bytes", int64(len(p)))
	return b.Backend.CompareAndSwap(c, k, t, p)
}
func (b *perfBackend) Sync(c context.Context, k string) error {
	b.add("backend_sync", 1)
	return b.Backend.(publication.DurabilityFence).Sync(c, k)
}

type perfSample struct {
	WallNS     int64            `json:"wall_ns"`
	CPUSeconds float64          `json:"cpu_seconds"`
	IO         map[string]int64 `json:"linux_process_io"`
	Counts     map[string]int64 `json:"counts"`
}

func perfNow(b *perfBackend) perfSample {
	var r syscall.Rusage
	if err := syscall.Getrusage(syscall.RUSAGE_SELF, &r); err != nil {
		panic(err)
	}
	p, e := os.ReadFile("/proc/self/io")
	if e != nil {
		panic(e)
	}
	m := map[string]int64{}
	for _, line := range strings.Split(string(p), "\n") {
		f := strings.Fields(line)
		if len(f) == 2 {
			n, e := strconv.ParseInt(f[1], 10, 64)
			if e != nil {
				panic(e)
			}
			m[strings.TrimSuffix(f[0], ":")] = n
		}
	}
	cpu := float64(r.Utime.Sec+r.Stime.Sec) + float64(r.Utime.Usec+r.Stime.Usec)/1e6
	return perfSample{time.Now().UnixNano(), cpu, m, b.snapshot()}
}
func perfDelta(a, z perfSample) perfSample {
	// Samples remain cumulative and immutable, including an ACK recorder's endpoint.
	// A struct copy alone still aliases these maps.
	difference := func(start, end map[string]int64) map[string]int64 {
		out := make(map[string]int64, len(start)+len(end))
		for k, v := range start {
			out[k] = -v
		}
		for k, v := range end {
			out[k] += v
		}
		return out
	}
	return perfSample{WallNS: z.WallNS - a.WallNS, CPUSeconds: z.CPUSeconds - a.CPUSeconds,
		IO: difference(a.IO, z.IO), Counts: difference(a.Counts, z.Counts)}
}

// Sequence and sample are published together only after sampling finishes. The
// backend ACK count may already be visible while a callback is still sampling;
// that count is never an endpoint-readiness signal.
type perfACKRecorder struct {
	mu       sync.Mutex
	sequence int64
	last     perfSample
}

func (a *perfACKRecorder) record(b *perfBackend, sample func() perfSample) {
	a.mu.Lock()
	defer a.mu.Unlock()
	b.add("fault_ack-committed", 1)
	s := sample()
	a.sequence++
	a.last = s
}
func (a *perfACKRecorder) endpoint(required int64) (perfSample, bool) {
	// A callback still sampling is not a published endpoint. Do not block the
	// observer or return the previous sample while that callback holds the lock.
	if !a.mu.TryLock() {
		return perfSample{}, false
	}
	defer a.mu.Unlock()
	return a.last, a.sequence >= required
}
func (a *perfACKRecorder) count() int64 {
	a.mu.Lock()
	defer a.mu.Unlock()
	return a.sequence
}

func TestPerfDeltaPreservesInputs(t *testing.T) {
	a := perfSample{WallNS: 10, CPUSeconds: 2, IO: map[string]int64{"read": 7, "only_start": 3}, Counts: map[string]int64{"ack": 4}}
	z := perfSample{WallNS: 40, CPUSeconds: 5, IO: map[string]int64{"read": 19, "only_end": 8}, Counts: map[string]int64{"ack": 6}}
	wantA := perfSample{WallNS: 10, CPUSeconds: 2, IO: map[string]int64{"read": 7, "only_start": 3}, Counts: map[string]int64{"ack": 4}}
	wantZ := perfSample{WallNS: 40, CPUSeconds: 5, IO: map[string]int64{"read": 19, "only_end": 8}, Counts: map[string]int64{"ack": 6}}
	want := perfSample{WallNS: 30, CPUSeconds: 3, IO: map[string]int64{"read": 12, "only_start": -3, "only_end": 8}, Counts: map[string]int64{"ack": 2}}
	d := perfDelta(a, z)
	if !reflect.DeepEqual(d, want) || !reflect.DeepEqual(a, wantA) || !reflect.DeepEqual(z, wantZ) {
		t.Fatal("delta mutated input or produced wrong result")
	}
	d.IO["read"] = 999
	d.Counts["ack"] = 999
	if !reflect.DeepEqual(a, wantA) || !reflect.DeepEqual(z, wantZ) {
		t.Fatal("delta maps alias inputs")
	}
	if again := perfDelta(a, z); !reflect.DeepEqual(again, want) {
		t.Fatal("repeated delta changed cumulative endpoint")
	}
}

func TestPerfACKEndpointWaitsForFinalSample(t *testing.T) {
	b := &perfBackend{counts: map[string]int64{}}
	var a perfACKRecorder
	a.record(b, func() perfSample { return perfSample{WallNS: 1, Counts: b.snapshot()} })
	// Two staged jobs require two published samples, not one ACK plus a
	// drained queue. The previous endpoint cannot satisfy the final boundary.
	if _, ready := a.endpoint(2); ready {
		t.Fatal("one ACK satisfied two staged jobs")
	}
	beforeCallback, beginCallback := make(chan struct{}), make(chan struct{})
	entered, release, finished := make(chan struct{}), make(chan struct{}), make(chan struct{})
	go func() {
		close(beforeCallback)
		<-beginCallback
		a.record(b, func() perfSample { close(entered); <-release; return perfSample{WallNS: 2, Counts: b.snapshot()} })
		close(finished)
	}()
	<-beforeCallback
	// Model the final DB commit becoming visible BEFORE its hook even begins.
	// The prior ACK sample exists and the queue is drained, but the exact two-job
	// boundary must still reject it. Testing only a delayed lock-holder misses this.
	drained := true
	_, beforeReady := a.endpoint(2)
	beforeCount := b.snapshot()["fault_ack-committed"]
	close(beginCallback)
	<-entered
	// The barrier models a drained work store and an already incremented ACK
	// hook count, with the FINAL sample still unavailable. No timing/sleep race.
	countVisible := b.snapshot()["fault_ack-committed"] == 2
	_, early := a.endpoint(2)
	close(release)
	<-finished
	if !drained || beforeCount != 1 || beforeReady {
		t.Fatal("drained queue returned previous sample before final callback began")
	}
	if !countVisible {
		t.Fatal("barrier did not expose old count/sample race")
	}
	if early {
		t.Fatal("endpoint returned before delayed final sample")
	}
	s, ready := a.endpoint(2)
	if !ready || s.WallNS != 2 || s.Counts["fault_ack-committed"] != 2 || a.count() != 2 {
		t.Fatal("final endpoint omitted staged ACK or used previous sample")
	}
}
func perfEmit(t *testing.T, phase string, a, z perfSample) {
	t.Helper()
	d := perfDelta(a, z)
	p, e := json.Marshal(struct {
		Phase               string     `json:"phase"`
		Profiled            bool       `json:"profiled"`
		Diagnostic          bool       `json:"diagnostic"`
		Sample              perfSample `json:"sample"`
		SettledTargetCPU    float64    `json:"settled_target_cpu_seconds"`
		AppendTargetSeconds int        `json:"append_target_seconds"`
	}{phase, os.Getenv("MEMORY_PERF_PROFILE") != "", os.Getenv("MEMORY_PERF_DIAGNOSTIC") == "1", d, 3, 10})
	if e != nil {
		t.Fatal(e)
	}
	t.Log("PERF_SCALAR " + string(p))
	if d.CPUSeconds > perfCPU {
		t.Fatal("phase CPU diagnostic limit exceeded")
	}
}
func perfSize(root string) (int64, error) {
	var n int64
	e := filepath.WalkDir(root, func(_ string, d fs.DirEntry, e error) error {
		if e != nil {
			return e
		}
		if d.Type().IsRegular() {
			s, e := d.Info()
			if e != nil {
				return e
			}
			n += s.Size()
		}
		return nil
	})
	return n, e
}
func perfWrite(t *testing.T, p string, b []byte) {
	t.Helper()
	if e := os.MkdirAll(filepath.Dir(p), 0700); e != nil {
		t.Fatal(e)
	}
	if e := os.WriteFile(p, b, 0600); e != nil {
		t.Fatal(e)
	}
}
func perfBody(n, seed int) []byte {
	if n > 4096 {
		var out []byte
		for i := 0; i < n/4096; i++ {
			out = append(out, perfBody(4096, seed*32+i)...)
		}
		return out
	}
	prefix := []byte("{\"type\":\"synthetic\",\"text\":\"")
	suffix := []byte("\"}\n")
	p := make([]byte, n)
	copy(p, prefix)
	x := uint64(seed + 1)
	for i := len(prefix); i < n-len(suffix); i++ {
		x ^= x << 13
		x ^= x >> 7
		x ^= x << 17
		p[i] = "abcdefghijklmnopqrstuvwxyz0123456789"[x%36]
	}
	copy(p[n-len(suffix):], suffix)
	return p
}
func perfHash(b []byte) string { s := sha256.Sum256(b); return hex.EncodeToString(s[:]) }

// ExportSessionBounded is the authoritative reader. No direct object decoding.
// Checks every expected source, not merely one readable main; rejects extras too.
func perfVerify(t *testing.T, root string, b publication.Backend, sources []perfSource) map[string]publication.Source {
	t.Helper()
	groups := map[publication.Identity][]perfSource{}
	for _, s := range sources {
		groups[s.req.Identity] = append(groups[s.req.Identity], s)
	}
	out := map[string]publication.Source{}
	n := 0
	for id, want := range groups {
		dest := filepath.Join(root, fmt.Sprintf("export-%04d", n))
		n++
		exp, e := publication.ExportSessionBounded(t.Context(), b, id, dest, 2<<20)
		if e != nil {
			t.Fatal(e)
		}
		if exp.Manifest.Identity != id || len(exp.Manifest.Sources) != len(want) || len(exp.Bodies) != len(want) {
			t.Fatal("reader identity/source coverage mismatch")
		}
		byID := map[string]perfSource{}
		for _, s := range want {
			byID[s.req.SourceID] = s
		}
		for _, s := range exp.Manifest.Sources {
			w, ok := byID[s.SourceID]
			if !ok || s.Path != w.req.Path || s.Withheld != nil || s.EpochID == "" || s.Revision == 0 {
				t.Fatal("unexpected source/epoch/path")
			}
			if s.Object.SHA256 != perfHash(w.body) || s.Object.Size != int64(len(w.body)) {
				t.Fatalf("exact synthetic body commitment mismatch transform=%s", w.req.Transform)
			}
			key := id.BaseKey() + "/" + s.SourceID
			if _, ok := out[key]; ok {
				t.Fatal("duplicate source identity")
			}
			out[key] = s
		}
		for _, body := range exp.Bodies {
			p, e := os.ReadFile(filepath.Join(dest, body.Path))
			if e != nil {
				t.Fatal(e)
			}
			w, ok := byID[body.SourceID]
			if !ok || !bytes.Equal(p, w.body) || body.SHA256 != perfHash(p) {
				t.Fatal("reader exact body mismatch")
			}
		}
		if e = os.RemoveAll(dest); e != nil {
			t.Fatal(e)
		}
	}
	return out
}

// capWriter leaves profiles private and fails at a fixed byte bound.
type perfCapWriter struct {
	f        *os.File
	n        int64
	exceeded bool
}

func (w *perfCapWriter) Write(p []byte) (int, error) {
	if w.n+int64(len(p)) > 64<<20 {
		w.exceeded = true
		return 0, errors.New("profile limit")
	}
	n, e := w.f.Write(p)
	w.n += int64(n)
	return n, e
}
func perfProfile(t *testing.T, phase string) func() {
	t.Helper()
	dir := os.Getenv("MEMORY_PERF_PROFILE")
	if dir == "" {
		return func() {}
	}
	f, e := os.OpenFile(filepath.Join(dir, phase+".pprof"), os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
	if e != nil {
		t.Fatal(e)
	}
	w := &perfCapWriter{f: f}
	if e = pprof.StartCPUProfile(w); e != nil {
		t.Fatal(e)
	}
	return func() {
		pprof.StopCPUProfile()
		if e := f.Close(); e != nil {
			t.Error(e)
		}
		if w.exceeded {
			t.Error("profile diagnostic limit")
		}
	}
}

// This fallback callback is opt-in diagnostic evidence, not primary timing.
// Production classify() checks all known classes BEFORE invoking it. Always return
// exactly its existing default FailureTransient, regardless of diagnostic content.
type perfFallbackReport struct {
	Diagnostic bool     `json:"diagnostic"`
	SQLiteCode *int     `json:"sqlite_code,omitempty"`
	Errno      *uint64  `json:"errno,omitempty"`
	Signatures []string `json:"signatures"`
}

func perfFallbackClassify(err error, report func(perfFallbackReport)) sessioncapture.FailureClass {
	r := perfFallbackReport{Diagnostic: true}
	var se *sqlite.Error
	if errors.As(err, &se) {
		code := se.Code()
		r.SQLiteCode = &code
	}
	var errno syscall.Errno
	if errors.As(err, &errno) {
		code := uint64(errno)
		r.Errno = &code
	}
	seen := map[string]bool{}
	for _, known := range []struct {
		err  error
		code string
	}{{context.Canceled, "context_canceled"}, {context.DeadlineExceeded, "context_deadline"}, {fs.ErrNotExist, "fs_not_exist"}, {fs.ErrPermission, "fs_permission"}, {adapters.ErrSQLiteRollbackJournal, "sqlite_rollback_journal"}} {
		if errors.Is(err, known.err) {
			seen[known.code] = true
		}
	}
	var visit func(error)
	visit = func(e error) {
		if e == nil {
			return
		}
		// Equality against fixed source signatures only. Never emit Error(), a
		// type name, wrapper text, coordinates, or dynamically matched substrings.
		code := ""
		switch e.Error() {
		case "sqlite authorized: invalid options":
			code = "sqlite_invalid_options"
		case "sqlite authorized: private empty output required":
			code = "sqlite_output_invalid"
		case "sqlite authorized: main identity changed":
			code = "sqlite_main_identity_changed"
		case "sqlite authorized: original identity changed":
			code = "sqlite_original_identity_changed"
		case "sqlite authorized: source changed":
			code = "sqlite_source_identity_changed"
		case "sqlite authorized: source bytes changed":
			code = "sqlite_source_bytes_changed"
		case "sqlite authorized: nonregular WAL":
			code = "sqlite_wal_nonregular"
		case "sqlite authorized: unresolved WAL entry":
			code = "sqlite_wal_unresolved"
		case "sqlite authorized: WAL alias changed":
			code = "sqlite_wal_alias_changed"
		case "sqlite authorized: sidecar parent changed":
			code = "sqlite_sidecar_parent_changed"
		case "sqlite authorized: WAL appeared":
			code = "sqlite_wal_appeared"
		case "sqlite authorized: source pair exceeds limit":
			code = "sqlite_source_pair_limit"
		case "sqlite authorized: identity metadata limit exceeded":
			code = "sqlite_identity_limit"
		case "sqlite authorized: excluded identity":
			code = "sqlite_identity_excluded"
		case "authorized sqlite VFS missing":
			code = "sqlite_vfs_missing"
		case "sqlite methods allocation failed":
			code = "sqlite_methods_allocation"
		case "sqlite backup unavailable", "sqlite driver has no backup API":
			code = "sqlite_driver_backup_unavailable"
		case "sqlite backup working limit exceeded":
			code = "sqlite_backup_working_limit"
		case "sessioncapture: source changed during observation":
			code = "capture_source_changed"
		case "sessioncapture: regular source required":
			code = "capture_nonregular_source"
		case "safefile: replaced or nonregular input":
			code = "safefile_replaced_or_nonregular"
		case "safefile: canonical absolute path required":
			code = "safefile_noncanonical_path"
		}
		if code != "" {
			seen[code] = true
		}
		if joined, ok := e.(interface{ Unwrap() []error }); ok {
			for _, child := range joined.Unwrap() {
				visit(child)
			}
		} else if wrapped, ok := e.(interface{ Unwrap() error }); ok {
			visit(wrapped.Unwrap())
		}
	}
	visit(err)
	for code := range seen {
		r.Signatures = append(r.Signatures, code)
	}
	if len(r.Signatures) == 0 {
		r.Signatures = []string{"unknown"}
	}
	slices.Sort(r.Signatures)
	report(r)
	return sessioncapture.FailureTransient
}

// Never return raw stored errors, coordinates or status structs in public logs.
func perfSafeCategory(value string) string {
	switch value {
	case "":
		return "none"
	case "transient", "destination", "policy-blocked", "integrity-error", "quota", "timeout", "absent", "shutdown":
		return value
	default:
		return "other"
	}
}
func perfSafeAgent(agent string) string {
	switch agent {
	case "claude", "codex", "pi":
		return agent
	default:
		return "other"
	}
}
func perfSafeStep(value string) (string, string) {
	if value == "" {
		return "none", "none"
	}
	worker, _, _ := strings.Cut(value, ":")
	switch worker {
	case "capture", "delivery", "discovery", "enrichment", "optional-repair":
	default:
		worker = "other"
	}
	code := "other"
	// These fixed signatures yield category codes only; never echo any substring.
	switch {
	case strings.Contains(value, "SQLITE_BUSY"):
		code = "sqlite_busy"
	case strings.Contains(value, "SQLITE_LOCKED"):
		code = "sqlite_locked"
	case strings.Contains(value, "SQLITE_INTERRUPT"):
		code = "sqlite_interrupt"
	case strings.Contains(value, "context deadline exceeded"):
		code = "deadline"
	case strings.Contains(value, "context canceled"):
		code = "canceled"
	}
	return worker, code
}

		defer tick.Stop()
		for {
			select {
			case <-exited:
				t.Fatalf("runtime exited: %v", runErr)
			case <-ctx.Done():
				t.Fatal("total diagnostic deadline")
			case <-tick.C:
			}
			now := perfNow(b)
			if now.CPUSeconds-phaseStart.CPUSeconds > perfCPU {
				t.Fatal("phase CPU diagnostic limit")
			}
			st, e := q.RuntimeStatus(ctx)
			if e != nil {
				t.Fatal(e)
			}
			if st.Quarantined != 0 || st.Paused || st.StepErrors != 0 {
				perfFailureDiagnostic(t, phase, st, q, ctx)
				t.Fatal("runtime failure/partial work")
			}
			if st.PendingPublications == 0 && (!requireDiscovery || (st.Sources == int64(len(sources)) && st.PendingSources == 0 && st.DiscoveryComplete)) {
				if endpoint, ready := acknowledgements.endpoint(requiredACK); ready {
					return endpoint
				}
			}
		}
	}
	activeEnd := wait(activeACKBoundary, false)
	if activeEnd.Counts["fault_ack-committed"] != activeACKBoundary {
		t.Fatal("active ACK endpoint does not include exactly all staged jobs")
	}
	finishProfile()
	perfEmit(t, "active_delivery_to_final_ack", phaseStart, activeEnd)
	stopRun()
	if e := <-done; e != nil {
		t.Fatal(e)
	}
	<-exited
	jobs, e = q.Status(ctx)
	if e != nil {
		t.Fatal(e)
	}
	for _, job := range jobs {
		key := jobKey{job.Identity, job.Sequence}
		if stagedJobs[key] {
			if job.State != "published" {
				t.Fatal("initial staged job was not acknowledged as published")
			}
			delete(stagedJobs, key)
		}
	}
	if len(stagedJobs) != 0 {
		t.Fatal("initial staged job disappeared")
	}
	phase = "discovery_warmup"
	done = make(chan error, 1)
	exited = make(chan struct{})
	runctx, stopRun = context.WithCancel(ctx)
	defer stopRun()
	go func() {
		runErr = q.Run(runctx, b, sessioncapture.RuntimeOptions{Classify: diagnosticClassify, DiscoverBatch: discover, PollInterval: time.Second, OperationTimeout: 10 * time.Second, CaptureWorkers: 1, DeliveryWorkers: 1, MaxAttempts: 2})
		done <- runErr
		close(exited)
	}()
	phaseStart = perfNow(b)
	_ = wait(activeACKBoundary, true)
	before := perfVerify(t, root, local, sources)
	phase = "settled"
	settledStart := perfNow(b)
	phaseStart = settledStart
	finishProfile = perfProfile(t, "settled")
	timer := time.NewTimer(30 * time.Second)
	tick := time.NewTicker(time.Second)
	settled := false
	for !settled {
		select {
		case <-exited:
			t.Fatalf("runtime exited: %v", runErr)
		case <-ctx.Done():
			t.Fatal("total diagnostic deadline")
		case <-timer.C:
			settled = true
		case <-tick.C:
			if perfNow(b).CPUSeconds-settledStart.CPUSeconds > perfCPU {
				t.Fatal("settled phase CPU diagnostic limit")
			}
			n, e := perfSize(root)
			if e != nil || n > perfFootprint {
				t.Fatal("fixture footprint diagnostic limit")
			}
			n, e = perfSize(c.Spool)
			if e != nil || n > perfSpool {
				t.Fatal("spool footprint diagnostic limit")
			}
		}
	}
	tick.Stop()
	settledEnd := perfNow(b)
	finishProfile()
	perfEmit(t, "settled_unchanged_30s", settledStart, settledEnd)
	if settledEnd.Counts["discovery_calls"] <= settledStart.Counts["discovery_calls"] {
		t.Fatal("no continuing periodic discovery")
	}
	for _, key := range []string{"backend_put", "backend_cas", "fault_upload-before", "fault_manifest-before", "fault_ack-committed"} {
		if settledEnd.Counts[key] != settledStart.Counts[key] {
			t.Fatalf("settled unexpected delivery operation: %s", key)
		}
	}
	after := perfVerify(t, root, local, sources)
	for k, s := range before {
		z := after[k]
		if s.SourceID != z.SourceID || s.EpochID != z.EpochID || s.Revision != z.Revision || s.Object != z.Object {
			t.Fatal("unchanged source advanced")
		}
	}
	// Append only raw mains, one per adapter. No Hint, WriteHint, hook or wake call.
	phase = "append"
	appendACKBoundary := acknowledgements.count() + 3
	phaseStart = perfNow(b)
	finishProfile = perfProfile(t, "append")
	for i := range sources {
		if sources[i].append {
			suffix := []byte("{\"type\":\"synthetic-append\"}\n")
			f, e := os.OpenFile(sources[i].req.Path, os.O_APPEND|os.O_WRONLY, 0600)
			if e != nil {
				t.Fatal(e)
			}
			_, e = f.Write(suffix)
			e = errors.Join(e, f.Sync(), f.Close())
			if e != nil {
				t.Fatal(e)
			}
			sources[i].body = append(sources[i].body, suffix...)
		}
	}
	appendEnd := wait(appendACKBoundary, true)
	if appendEnd.Counts["fault_ack-committed"] != appendACKBoundary {
		t.Fatal("append ACK endpoint boundary mismatch")
	}
	finishProfile()
	perfEmit(t, "unhinted_append_to_final_ack", phaseStart, appendEnd)
	after = perfVerify(t, root, local, sources)
	affected := 0
	for _, w := range sources {
		k := w.req.Identity.BaseKey() + "/" + w.req.SourceID
		a, z := before[k], after[k]
		if a.SourceID != z.SourceID || a.EpochID != z.EpochID {
			t.Fatal("append changed source/epoch identity")
		}
		if w.append {
			affected++
			if z.Revision != a.Revision+1 {
				t.Fatal("affected revision not exactly +1")
			}
		} else if z.Revision != a.Revision || z.Object != a.Object {
			t.Fatal("unrelated source changed")
		}
	}
	if affected != 3 {
		t.Fatal("append fixture coverage")
	}
	n, e := perfSize(root)
	if e != nil || n > perfFootprint {
		t.Fatal("fixture footprint diagnostic limit")
	}
	t.Logf("PERF_CORRECTNESS ordinary_mains=96 artifacts=96 compressed=3 configs=3 sqlite=3 sources=%d changed=3 exact_reader_hashes=true stable_epochs=true unrelated_unchanged=true credential_negatives=3 fixture_bytes=%d", len(sources), n)
}
